cmake_minimum_required(VERSION 3.20)
project(devshell_cgal_worker LANGUAGES CXX)
find_package(CGAL REQUIRED)
add_executable(devshell_cgal_worker main.cpp)
target_compile_features(devshell_cgal_worker PRIVATE cxx_std_17)
target_link_libraries(devshell_cgal_worker PRIVATE CGAL::CGAL)
if(MSVC)
  target_compile_options(devshell_cgal_worker PRIVATE /bigobj /utf-8)
endif()
